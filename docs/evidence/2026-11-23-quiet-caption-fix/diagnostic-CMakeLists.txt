cmake_minimum_required(VERSION 3.20)
project(CaptionBaselineDiagnostic LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
find_package(Baffle 0.1.0 EXACT CONFIG REQUIRED)
add_subdirectory(../../examples/win32-transcript app)
get_target_property(links BaffleLive LINK_LIBRARIES)
get_target_property(includes BaffleLive INCLUDE_DIRECTORIES)
add_executable(ReplayBaseline ../../examples/win32-transcript/replay_test.c
               ../vad/speech.c ../../examples/win32-transcript/captions.c)
target_compile_definitions(ReplayBaseline PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
 BF_REPLAY_APP_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../baseline/app.c")
target_compile_options(ReplayBaseline PRIVATE /W4 /WX /utf-8 /fp:precise)
target_include_directories(ReplayBaseline PRIVATE ${includes} ../../examples/win32-transcript)
target_link_libraries(ReplayBaseline PRIVATE ${links})
add_executable(activity_bench activity_bench.c)
target_link_libraries(activity_bench PRIVATE Baffle::baffle)
