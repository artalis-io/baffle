cmake_minimum_required(VERSION 3.20)
project(BaffleAcousticRunner LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# External qualification only: installed public Baffle, optional external AEC3.
find_package(Baffle 0.1.0 EXACT CONFIG REQUIRED)
option(BF_COMPARE_WEBRTC "Compare with installed WebRTC AudioProcessing >=2.1" OFF)
option(BF_COMPARE_WEBRTC_LINEAR "Enable diagnostic export against the separately patched 2.1 package" OFF)
option(BF_QUALIFY_HAIL "Build installed-public-Hail raw acquisition tool" OFF)
add_executable(baffle_vad_trace vad_trace.c)
target_compile_features(baffle_vad_trace PRIVATE c_std_11)
target_link_libraries(baffle_vad_trace PRIVATE Baffle::baffle)
if(BF_QUALIFY_HAIL)
  find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
  add_executable(baffle_hail_acquire hail_acquire.c)
  target_compile_features(baffle_hail_acquire PRIVATE c_std_11)
  target_link_libraries(baffle_hail_acquire PRIVATE Hail::hail)
  add_executable(baffle_live_capture live_capture.c)
  target_compile_features(baffle_live_capture PRIVATE c_std_11)
  target_link_libraries(baffle_live_capture PRIVATE Hail::hail Baffle::baffle)
  if(MSVC)
    target_compile_options(baffle_live_capture PRIVATE /W4 /WX)
  else()
    target_compile_options(baffle_live_capture PRIVATE -Wall -Wextra -Werror)
  endif()
  enable_testing()
  add_executable(baffle_hail_retirement_test retirement_test.c)
  target_compile_features(baffle_hail_retirement_test PRIVATE c_std_11)
  target_link_libraries(baffle_hail_retirement_test PRIVATE Hail::hail)
  add_test(NAME hail_retirement_ownership COMMAND baffle_hail_retirement_test)
endif()
add_executable(baffle_acoustic_runner main.c)
add_executable(baffle_residual_runner main.c residual_experiment.c)
target_compile_features(baffle_residual_runner PRIVATE c_std_11)
target_compile_definitions(baffle_residual_runner PRIVATE BF_EXPERIMENTAL_RESIDUAL=1)
target_link_libraries(baffle_residual_runner PRIVATE Baffle::baffle ${CMAKE_DL_LIBS})
add_executable(baffle_noise_order noise_order.c)
target_compile_features(baffle_noise_order PRIVATE c_std_11)
target_link_libraries(baffle_noise_order PRIVATE Baffle::baffle)
target_compile_features(baffle_acoustic_runner PRIVATE c_std_11)
target_link_libraries(baffle_acoustic_runner PRIVATE Baffle::baffle)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(baffle_acoustic_runner PRIVATE ${CMAKE_DL_LIBS})
  enable_language(C)
  add_library(baffle_qa_alloc_probe SHARED allocation_probe.c)
  target_compile_features(baffle_qa_alloc_probe PRIVATE c_std_11)
endif()
if(BF_COMPARE_WEBRTC)
  enable_language(CXX)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WEBRTC REQUIRED IMPORTED_TARGET webrtc-audio-processing-2>=2.1)
  add_library(baffle_qa_webrtc STATIC webrtc_bridge.cpp)
  target_compile_features(baffle_qa_webrtc PRIVATE cxx_std_17)
  target_link_libraries(baffle_qa_webrtc PRIVATE PkgConfig::WEBRTC)
  target_link_libraries(baffle_acoustic_runner PRIVATE baffle_qa_webrtc)
  target_compile_definitions(baffle_acoustic_runner PRIVATE BF_COMPARE_WEBRTC=1)
  if(BF_COMPARE_WEBRTC_LINEAR)
    target_compile_definitions(baffle_acoustic_runner PRIVATE BF_COMPARE_WEBRTC_LINEAR=1)
    target_compile_definitions(baffle_qa_webrtc PRIVATE BF_COMPARE_WEBRTC_LINEAR=1)
  endif()
endif()
if(MSVC)
  target_compile_options(baffle_acoustic_runner PRIVATE /W4 /WX)
  target_compile_options(baffle_noise_order PRIVATE /W4 /WX)
  target_compile_options(baffle_residual_runner PRIVATE /W4 /WX)
else()
  target_compile_options(baffle_acoustic_runner PRIVATE -Wall -Wextra -Werror)
  target_compile_options(baffle_noise_order PRIVATE -Wall -Wextra -Werror)
  target_compile_options(baffle_residual_runner PRIVATE -Wall -Wextra -Werror)
endif()
