cmake_minimum_required(VERSION 3.20)
project(BaffleInstalledConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BF_TEST_CXX_COMPAT "Exercise a foreign C++ caller of the public C ABI" OFF)
find_package(Baffle 0.1.0 EXACT CONFIG REQUIRED)
enable_testing()
add_executable(consumer_c main.c)
set(BF_CONSUMERS consumer_c)
if(BF_TEST_CXX_COMPAT)
  enable_language(CXX)
  add_executable(consumer_cpp main.cpp)
  target_compile_features(consumer_cpp PRIVATE cxx_std_11)
  list(APPEND BF_CONSUMERS consumer_cpp)
endif()
foreach(target IN LISTS BF_CONSUMERS)
  target_link_libraries(${target} PRIVATE Baffle::baffle)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
