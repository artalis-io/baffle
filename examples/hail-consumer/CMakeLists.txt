cmake_minimum_required(VERSION 3.20)
project(BaffleHailInstalledConsumer LANGUAGES C CXX)
find_package(BaffleHail 0.1.0 EXACT CONFIG REQUIRED)
add_executable(baffle_hail_consumer main.c)
target_link_libraries(baffle_hail_consumer PRIVATE Baffle::hail_adapter)
target_compile_features(baffle_hail_consumer PRIVATE c_std_11)
if(MSVC)
  target_compile_options(baffle_hail_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(baffle_hail_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME installed_hail_duplex COMMAND baffle_hail_consumer)
set_tests_properties(installed_hail_duplex PROPERTIES TIMEOUT 180)
find_package(Threads REQUIRED)
add_executable(baffle_hail_concurrent concurrent.cpp)
target_compile_features(baffle_hail_concurrent PRIVATE cxx_std_11)
target_link_libraries(baffle_hail_concurrent PRIVATE Baffle::hail_adapter Threads::Threads)
add_test(NAME installed_hail_concurrent COMMAND baffle_hail_concurrent)
set_tests_properties(installed_hail_concurrent PROPERTIES TIMEOUT 180)
add_executable(baffle_hail_minimal minimal.c)
target_compile_features(baffle_hail_minimal PRIVATE c_std_11)
set_target_properties(baffle_hail_minimal PROPERTIES C_EXTENSIONS OFF)
target_link_libraries(baffle_hail_minimal PRIVATE Baffle::hail_adapter)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(baffle_hail_minimal PRIVATE -Wl,--as-needed)
endif()
add_test(NAME installed_hail_minimal COMMAND baffle_hail_minimal)
