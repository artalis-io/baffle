cmake_minimum_required(VERSION 3.20)
project(BaffleHailInstalledConsumer LANGUAGES C)
find_package(BaffleHail 0.1.0 EXACT CONFIG REQUIRED)
add_executable(baffle_hail_consumer main.c)
target_link_libraries(baffle_hail_consumer PRIVATE Baffle::hail_adapter)
target_compile_features(baffle_hail_consumer PRIVATE c_std_11)
if(MSVC)
  target_compile_options(baffle_hail_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(baffle_hail_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME installed_hail_duplex COMMAND baffle_hail_consumer)
set_tests_properties(installed_hail_duplex PROPERTIES TIMEOUT 180)
