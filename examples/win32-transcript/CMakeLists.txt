cmake_minimum_required(VERSION 3.20)
project(BaffleWin32Transcript LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT WIN32)
  message(FATAL_ERROR "This external demonstration requires Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
find_package(Baffle 0.1.0 EXACT CONFIG REQUIRED)
set(SHERPA_SOURCE_DIR "" CACHE PATH "sherpa-onnx v1.12.40 source headers")
set(SHERPA_STATIC_DIR "" CACHE PATH "Matching Windows x64 static MT no-TTS library package")
if(NOT EXISTS "${SHERPA_SOURCE_DIR}/sherpa-onnx/c-api/c-api.h")
  message(FATAL_ERROR "Supply matching sherpa-onnx source headers")
endif()
add_executable(BaffleLive WIN32 main.c app.c speech.c captions.c ../../tools/acoustic-runner/live_capture.c)
target_compile_features(BaffleLive PRIVATE c_std_11)
target_compile_definitions(BaffleLive PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN BF_LIVE_SOURCE_ONLY)
target_compile_options(BaffleLive PRIVATE /W4 /WX /utf-8 /fp:precise)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
  target_compile_options(BaffleLive PRIVATE /experimental:c11atomics)
endif()
target_include_directories(BaffleLive PRIVATE "${SHERPA_SOURCE_DIR}" ../../tools/acoustic-runner)
target_link_libraries(BaffleLive PRIVATE Hail::hail Baffle::baffle user32 gdi32 comctl32 uxtheme dwmapi shell32 shlwapi bcrypt ws2_32 dbghelp advapi32 runtimeobject)
foreach(name sherpa-onnx-c-api sherpa-onnx-core sherpa-onnx-kaldifst-core sherpa-onnx-fstfar sherpa-onnx-fst
             kaldi-decoder-core kaldi-native-fbank-core kissfft-float ssentencepiece_core onnxruntime)
  if(NOT EXISTS "${SHERPA_STATIC_DIR}/lib/${name}.lib")
    message(FATAL_ERROR "Missing static library ${name}")
  endif()
  target_link_libraries(BaffleLive PRIVATE "${SHERPA_STATIC_DIR}/lib/${name}.lib")
endforeach()
target_link_options(BaffleLive PRIVATE /OPT:REF /OPT:ICF)
install(TARGETS BaffleLive RUNTIME DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(caption_tests caption_tests.c captions.c)
  target_compile_features(caption_tests PRIVATE c_std_11)
  target_compile_options(caption_tests PRIVATE /W4 /WX /utf-8)
  add_test(NAME caption_boundaries COMMAND caption_tests)
  add_test(NAME spsc_integrity COMMAND BaffleLive --queue-test)
  add_test(NAME wav_boundary_validation COMMAND BaffleLive --wave-test)
endif()

# Optional file-fed application diagnostic: no microphone, devices or cloud.
option(BF_BUILD_CAPTION_REPLAY "Build the file-fed real-time caption diagnostic" OFF)
if(BF_BUILD_CAPTION_REPLAY)
  add_executable(BaffleReplay replay_test.c speech.c captions.c)
  target_compile_features(BaffleReplay PRIVATE c_std_11)
  target_compile_definitions(BaffleReplay PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(BaffleReplay PRIVATE /W4 /WX /utf-8 /fp:precise)
  if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(BaffleReplay PRIVATE /experimental:c11atomics)
  endif()
  get_target_property(BF_REPLAY_LINKS BaffleLive LINK_LIBRARIES)
  target_link_libraries(BaffleReplay PRIVATE ${BF_REPLAY_LINKS})
  target_include_directories(BaffleReplay PRIVATE "${SHERPA_SOURCE_DIR}")
endif()
