cmake_minimum_required(VERSION 3.20)
project(Baffle VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(BF_BUILD_SHARED "Build shared Baffle" OFF)
option(BF_BUILD_TESTS "Build deterministic tests" ON)
option(BF_BUILD_BENCHMARKS "Build standalone CPU benchmarks" ON)
option(BF_BUILD_TOOLS "Build offline WAV qualification tool" OFF)
option(BF_WITH_SPEEXDSP "Enable pinned isolated SpeexDSP AEC backend" OFF)
option(BF_WITH_HAIL "Build independent installed-Hail adapter" OFF)
option(BF_BUILD_FUZZER "Build Clang libFuzzer harness" OFF)
set(BF_SANITIZER "" CACHE STRING "address, undefined, thread (GCC/Clang)")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BF_KIND STATIC)
if(BF_BUILD_SHARED)
  set(BF_KIND SHARED)
endif()
add_library(baffle ${BF_KIND} src/core.c src/pcm.c src/clock.c src/resampler.c src/channels.c src/conditioning.c src/noise.c src/aec.c src/pipeline.c)
add_library(Baffle::baffle ALIAS baffle)
target_compile_features(baffle PUBLIC c_std_11)
target_include_directories(baffle PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(baffle PROPERTIES C_EXTENSIONS OFF C_VISIBILITY_PRESET hidden VERSION ${PROJECT_VERSION} SOVERSION 0)
target_compile_definitions(baffle PRIVATE BF_BUILDING_LIBRARY)
if(NOT BF_BUILD_SHARED)
  target_compile_definitions(baffle PUBLIC BF_STATIC)
endif()
if(MSVC)
  target_compile_options(baffle PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(baffle PRIVATE -Wall -Wextra -Wpedantic -Werror -Wshadow -Wconversion -fno-fast-math)
  target_link_libraries(baffle PUBLIC m)
endif()
if(BF_SANITIZER)
  if(MSVC)
    message(FATAL_ERROR "Use GCC/Clang driver sanitizer builds; Windows ClangCL ASan is not configured")
  endif()
  if(BF_SANITIZER STREQUAL "address")
    set(BF_SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(BF_SANITIZER STREQUAL "thread")
    set(BF_SAN_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  elseif(BF_SANITIZER STREQUAL "undefined")
    set(BF_SAN_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "Unknown sanitizer")
  endif()
  target_compile_options(baffle PUBLIC ${BF_SAN_FLAGS})
  target_link_options(baffle PUBLIC ${BF_SAN_FLAGS})
endif()
target_compile_definitions(baffle PRIVATE BF_HAVE_SPEEXDSP=$<BOOL:${BF_WITH_SPEEXDSP}>)
if(BF_WITH_SPEEXDSP)
  include(cmake/SpeexDSP.cmake)
endif()
if(BF_BUILD_TESTS)
  enable_testing()
  add_executable(baffle_conformance tests/conformance.c)
  target_link_libraries(baffle_conformance PRIVATE Baffle::baffle)
  add_test(NAME conformance COMMAND baffle_conformance)
  add_executable(baffle_pcm tests/pcm.c)
  target_link_libraries(baffle_pcm PRIVATE Baffle::baffle)
  add_test(NAME pcm COMMAND baffle_pcm)
  add_executable(baffle_numerical tests/numerical.c)
  target_link_libraries(baffle_numerical PRIVATE Baffle::baffle)
  add_test(NAME numerical COMMAND baffle_numerical)
  add_executable(baffle_conditioning tests/conditioning.c)
  target_link_libraries(baffle_conditioning PRIVATE Baffle::baffle)
  add_test(NAME conditioning COMMAND baffle_conditioning)
  add_executable(baffle_aec tests/aec.c)
  target_link_libraries(baffle_aec PRIVATE Baffle::baffle)
  add_test(NAME aec COMMAND baffle_aec)
  add_executable(baffle_pipeline tests/pipeline.c)
  target_link_libraries(baffle_pipeline PRIVATE Baffle::baffle)
  add_test(NAME pipeline COMMAND baffle_pipeline)
  enable_language(CXX)
  find_package(Threads REQUIRED)
  add_executable(baffle_concurrency tests/concurrency.cpp)
  target_compile_features(baffle_concurrency PRIVATE cxx_std_11)
  target_link_libraries(baffle_concurrency PRIVATE Baffle::baffle Threads::Threads)
  add_test(NAME concurrency COMMAND baffle_concurrency)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT BF_BUILD_SHARED)
    add_executable(baffle_noalloc tests/noalloc.c)
    target_link_libraries(baffle_noalloc PRIVATE Baffle::baffle)
    target_link_options(baffle_noalloc PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
    add_test(NAME noalloc COMMAND baffle_noalloc)
  endif()
endif()
if(BF_BUILD_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR MSVC)
    message(FATAL_ERROR "libFuzzer requires the Clang driver")
  endif()
  target_compile_options(baffle PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(baffle PUBLIC -fsanitize=fuzzer-no-link,address,undefined)
  if(BF_WITH_SPEEXDSP)
    target_compile_options(baffle_speex_kernel PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  endif()
  foreach(name transport pipeline aec)
    add_executable(baffle_fuzz_${name} fuzz/${name}.c)
    target_link_libraries(baffle_fuzz_${name} PRIVATE Baffle::baffle)
    target_compile_options(baffle_fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(baffle_fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()
if(BF_BUILD_BENCHMARKS)
  add_executable(baffle_latency bench/latency.c)
  target_link_libraries(baffle_latency PRIVATE Baffle::baffle)
endif()
if(BF_BUILD_TOOLS)
  add_executable(baffle_qualify tools/qualify.c)
  target_link_libraries(baffle_qualify PRIVATE Baffle::baffle)
endif()
install(TARGETS baffle EXPORT BaffleTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/baffle.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/Baffle)
install(EXPORT BaffleTargets NAMESPACE Baffle:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Baffle)
configure_package_config_file(cmake/BaffleConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BaffleConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Baffle)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BaffleConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BaffleConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BaffleConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Baffle)
set(BF_PC_DEFINES "")
set(BF_PC_LIBM "-lm")
if(MSVC)
  set(BF_PC_LIBM "")
endif()
if(NOT BF_BUILD_SHARED)
  set(BF_PC_DEFINES "-DBF_STATIC")
endif()
configure_file(cmake/baffle.pc.in baffle.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/baffle.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(BF_WITH_HAIL)
  find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
  add_library(baffle_hail_adapter ${BF_KIND} adapters/hail.c)
  add_library(Baffle::hail_adapter ALIAS baffle_hail_adapter)
  target_link_libraries(baffle_hail_adapter PUBLIC Baffle::baffle Hail::hail)
  target_compile_definitions(baffle_hail_adapter PRIVATE BF_BUILDING_HAIL_ADAPTER)
  if(NOT BF_BUILD_SHARED)
    target_compile_definitions(baffle_hail_adapter PUBLIC BF_HAIL_STATIC)
  endif()
  target_compile_features(baffle_hail_adapter PRIVATE c_std_11)
  set_target_properties(baffle_hail_adapter PROPERTIES C_EXTENSIONS OFF C_VISIBILITY_PRESET hidden)
  # Adapter-only consumers can drop a direct core dependency with --as-needed.
  # The installed adapter must locate its co-located core transitively.
  if(APPLE)
    set_target_properties(baffle_hail_adapter PROPERTIES INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_target_properties(baffle_hail_adapter PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  if(MSVC)
    target_compile_options(baffle_hail_adapter PRIVATE /W4 /WX)
  else()
    target_compile_options(baffle_hail_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  install(TARGETS baffle_hail_adapter EXPORT BaffleHailTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES include/baffle/hail.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/baffle)
  install(EXPORT BaffleHailTargets NAMESPACE Baffle:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BaffleHail)
  set_target_properties(baffle_hail_adapter PROPERTIES EXPORT_NAME hail_adapter)
  configure_package_config_file(cmake/BaffleHailConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BaffleHailConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BaffleHail)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BaffleHailConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BaffleHail)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BaffleConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BaffleHail RENAME BaffleHailConfigVersion.cmake)
endif()
